% 설치된 글꼴 이름으로 선택하므로 Windows/macOS/Linux에서 같은 파일을 사용한다.
% 우선 KoPubWorld, 미설치 시 시스템 한글 글꼴로 대체한다.
\ifproceedings
\IfFontExistsTF{Liberation Serif}{\setmainfont{Liberation Serif}}{\IfFontExistsTF{Times New Roman}{\setmainfont{Times New Roman}}{\setmainfont{Latin Modern Roman}}}
\IfFontExistsTF{Liberation Sans}{\setsansfont{Liberation Sans}}{\IfFontExistsTF{Arial}{\setsansfont{Arial}}{\setsansfont{Latin Modern Sans}}}
\else
\IfFontExistsTF{Liberation Serif}{\setmainfont[FakeStretch=.95,LetterSpace=-2.5]{Liberation Serif}}{\IfFontExistsTF{Times New Roman}{\setmainfont[FakeStretch=.95,LetterSpace=-2.5]{Times New Roman}}{\setmainfont[FakeStretch=.95,LetterSpace=-2.5]{Latin Modern Roman}}}
\IfFontExistsTF{Liberation Sans}{\setsansfont[FakeStretch=.95,LetterSpace=-2.5]{Liberation Sans}}{\IfFontExistsTF{Arial}{\setsansfont[FakeStretch=.95,LetterSpace=-2.5]{Arial}}{\setsansfont[FakeStretch=.95,LetterSpace=-2.5]{Latin Modern Sans}}}
\fi
\IfFontExistsTF{KoPubWorldBatang Medium}{
  \def\paperHangulMain{KoPubWorldBatang Medium}
  \def\paperHangulBold{KoPubWorldBatang Bold}
  \def\paperHangulSans{KoPubWorldDotum Medium}
  \def\paperHangulSansBold{KoPubWorldDotum Bold}
}{\IfFontExistsTF{Noto Serif CJK KR}{
  \def\paperHangulMain{Noto Serif CJK KR}
  \def\paperHangulBold{Noto Serif CJK KR Bold}
  \def\paperHangulSans{Noto Sans CJK KR}
  \def\paperHangulSansBold{Noto Sans CJK KR Bold}
}{\IfFontExistsTF{UnBatang}{
  \def\paperHangulMain{UnBatang}
  \def\paperHangulBold{UnBatang Bold}
  \def\paperHangulSans{UnDotum}
  \def\paperHangulSansBold{UnDotum Bold}
}{\IfFontExistsTF{Batang}{
  \def\paperHangulMain{Batang}
  \def\paperHangulBold{Batang}
  \def\paperHangulSans{Malgun Gothic}
  \def\paperHangulSansBold{Malgun Gothic Bold}
}{\IfFontExistsTF{Malgun Gothic}{
  \def\paperHangulMain{Malgun Gothic}
  \def\paperHangulBold{Malgun Gothic Bold}
  \def\paperHangulSans{Malgun Gothic}
  \def\paperHangulSansBold{Malgun Gothic Bold}
}{
  \PackageError{ieie-paper}{No Korean font found}{Install KoPubWorld Batang and Dotum, or Noto CJK Korean fonts; then restart the editor.}
}}}}}
\ifproceedings
\setmainhangulfont[BoldFont=\paperHangulBold]{\paperHangulMain}
\setsanshangulfont[BoldFont=\paperHangulSansBold]{\paperHangulSans}
\else
\setmainhangulfont[BoldFont=\paperHangulBold,FakeStretch=.95,LetterSpace=-2.5]{\paperHangulMain}
\setsanshangulfont[BoldFont=\paperHangulSansBold,FakeStretch=.95,LetterSpace=-2.5]{\paperHangulSans}
\fi
